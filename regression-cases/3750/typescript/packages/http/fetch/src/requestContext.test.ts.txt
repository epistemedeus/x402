import { describe, expect, it, vi } from "vitest";
import { x402Client, x402HTTPClient } from "@x402/core/client";
import { encodePaymentRequiredHeader } from "@x402/core/http";
import type { Network, PaymentRequired } from "@x402/core/types";
import type { SchemeNetworkClient } from "@x402/core/types";
import { wrapFetchWithPayment } from "./index";
import { httpPaymentCall } from "./requestCall";

const network = "test:network" as Network;

function required(): PaymentRequired {
  return {
    x402Version: 2,
    resource: { url: "https://pay.example/resource", description: "Test", mimeType: "text/plain" },
    accepts: [
      {
        scheme: "test-scheme",
        network,
        amount: "1000000",
        asset: "TEST",
        payTo: "test_recipient",
        maxTimeoutSeconds: 30,
        extra: {},
      },
    ],
  };
}

function paymentRequiredResponse(): Response {
  return new Response(null, {
    status: 402,
    headers: { "PAYMENT-REQUIRED": encodePaymentRequiredHeader(required()) },
  });
}

describe("fetch request snapshot", () => {
  it("does not read a body until getBody is called, then reads the stream once", async () => {
    let pulls = 0;
    const stream = new ReadableStream({
      pull(controller) {
        pulls += 1;
        controller.enqueue(new TextEncoder().encode("hello"));
        controller.close();
      },
    });
    const request = new Request("https://pay.example/stream", {
      method: "POST",
      body: stream,
      duplex: "half",
      headers: { "X-Request-Id": "stream", "Payment-Signature": "sig" },
    } as RequestInit);
    let bodySource: Request | undefined;
    const call = httpPaymentCall(request, () => {
      bodySource = request.clone();
      return bodySource;
    });

    expect(call.call.request.method).toBe("POST");
    expect(call.call.request.url).toBe("https://pay.example/stream");
    expect(call.call.request.headers["x-request-id"]).toBe("stream");
    expect(call.call.request.headers["payment-signature"]).toBeUndefined();
    expect(bodySource).toBeUndefined();
    expect(request.bodyUsed).toBe(false);
    expect(pulls).toBe(0);

    const first = await call.call.request.getBody?.();
    const second = await call.call.request.getBody?.();
    expect(pulls).toBe(1);
    expect(bodySource?.bodyUsed).toBe(true);
    expect(request.bodyUsed).toBe(false);
    expect(first).toEqual(new TextEncoder().encode("hello"));
    expect(second).toEqual(first);
    if (first && second) {
      first[0] = 0;
      expect(second[0]).not.toBe(0);
    }
  });

  it("omits getBody for a bodyless request", () => {
    const request = new Request("https://pay.example/get");
    const call = httpPaymentCall(request, () => request.clone());
    expect(call.call.request.method).toBe("GET");
    expect(call.call.request.getBody).toBeUndefined();
  });

  it("rejects an aborted or already used body without inventing bytes", async () => {
    const controller = new AbortController();
    const aborted = new Request("https://pay.example/abort", {
      method: "POST",
      body: "abc",
      signal: controller.signal,
    });
    const abortedBody = aborted.clone();
    controller.abort();
    const abortedCall = httpPaymentCall(aborted, () => abortedBody);
    await expect(abortedCall.call.request.getBody?.()).rejects.toBeTruthy();
    await expect(abortedCall.call.request.getBody?.()).rejects.toBeTruthy();

    const used = new Request("https://pay.example/used", { method: "POST", body: "abc" });
    const usedBody = used.clone();
    await usedBody.arrayBuffer();
    const usedCall = httpPaymentCall(used, () => usedBody);
    await expect(usedCall.call.request.getBody?.()).rejects.toThrow("request_body_already_used");
  });

  it("rejects a broken stream once and does not reread it", async () => {
    let pulls = 0;
    const stream = new ReadableStream({
      pull(controller) {
        pulls += 1;
        controller.error(new Error("stream_broken"));
      },
    });
    const request = new Request("https://pay.example/broken", {
      method: "POST",
      body: stream,
      duplex: "half",
    } as RequestInit);
    const call = httpPaymentCall(request, () => request.clone());
    await expect(call.call.request.getBody?.()).rejects.toThrow("stream_broken");
    await expect(call.call.request.getBody?.()).rejects.toThrow("stream_broken");
    expect(pulls).toBe(1);
  });
});

describe("wrapFetchWithPayment request context", () => {
  it("propagates the original call through first payment and recovery", async () => {
    const seen: Array<{ url: string; signature?: string; body?: string }> = [];
    const scheme: SchemeNetworkClient = {
      scheme: "test-scheme",
      async createPaymentPayload(_version, requirements, context) {
        expect(requirements.payTo).toBe("test_recipient");
        if (context?.call?.transport !== "http") {
          throw new Error("request_context_missing");
        }
        const body = context.call.request.getBody
          ? new TextDecoder().decode(await context.call.request.getBody())
          : undefined;
        seen.push({
          url: context.call.request.url,
          signature: context.call.request.headers["payment-signature"],
          body,
        });
        return { x402Version: 2, payload: { signature: "scheme_signature" } };
      },
    };
    const client = new x402Client().register(network, scheme).setSpendControls(false);
    let recovered = false;
    client.onPaymentResponse(async () => {
      if (recovered) return;
      recovered = true;
      return { recovered: true };
    });
    const http = new x402HTTPClient(client);
    let attempts = 0;
    const bodies: string[] = [];
    const fetchMock = vi.fn(async (input: RequestInfo | URL) => {
      const request = input instanceof Request ? input : new Request(input);
      attempts += 1;
      bodies.push(await request.clone().text());
      if (attempts < 3) return paymentRequiredResponse();
      return new Response("ok", { status: 200 });
    });
    const paid = wrapFetchWithPayment(fetchMock, http);
    const response = await paid("https://pay.example/recover", {
      method: "POST",
      body: "payload-bytes",
      headers: { "X-Request-Id": "recover" },
    });

    expect(response.status).toBe(200);
    expect(seen).toEqual([
      { url: "https://pay.example/recover", signature: undefined, body: "payload-bytes" },
      { url: "https://pay.example/recover", signature: undefined, body: "payload-bytes" },
    ]);
    expect(bodies).toEqual(["payload-bytes", "payload-bytes", "payload-bytes"]);
    const paidRequest = fetchMock.mock.calls[1][0] as Request;
    expect(paidRequest.headers.get("PAYMENT-SIGNATURE")).toBeTruthy();
    expect(paidRequest.headers.get("x-request-id")).toBe("recover");
  });

  it("keeps concurrent fetches on one client bound to their own urls", async () => {
    let release: () => void = () => {};
    const gate = new Promise<void>(resolve => {
      release = resolve;
    });
    let waiting = 0;
    const seen: string[] = [];
    const scheme: SchemeNetworkClient = {
      scheme: "test-scheme",
      async createPaymentPayload(_version, _requirements, context) {
        if (context?.call?.transport !== "http") throw new Error("request_context_missing");
        const url = context.call.request.url;
        waiting += 1;
        await gate;
        seen.push(url);
        return { x402Version: 2, payload: { signature: "scheme_signature" } };
      },
    };
    const client = new x402Client().register(network, scheme).setSpendControls(false);
    const paid = wrapFetchWithPayment(async request => {
      const current = request as Request;
      if (!current.headers.get("PAYMENT-SIGNATURE")) return paymentRequiredResponse();
      return new Response(current.url, { status: 200 });
    }, client);

    const first = paid("https://pay.example/a");
    const second = paid("https://pay.example/b");
    const started = Date.now();
    while (waiting < 2) {
      if (Date.now() - started > 1000) throw new Error("timed out waiting for concurrent fetches");
      await new Promise(resolve => setTimeout(resolve, 5));
    }
    release();
    const results = await Promise.all([first, second]);
    expect(seen.sort()).toEqual(["https://pay.example/a", "https://pay.example/b"]);
    expect((await results[0].text()).startsWith("https://pay.example/")).toBe(true);
  });

  it("fails closed when the body stream aborts instead of signing empty bytes", async () => {
    let signed = false;
    const scheme: SchemeNetworkClient = {
      scheme: "test-scheme",
      async createPaymentPayload(_version, _requirements, context) {
        if (context?.call?.transport !== "http" || !context.call.request.getBody) {
          throw new Error("request_context_missing");
        }
        await context.call.request.getBody();
        signed = true;
        return { x402Version: 2, payload: { signature: "scheme_signature" } };
      },
    };
    const client = new x402Client().register(network, scheme).setSpendControls(false);
    const controller = new AbortController();
    const paid = wrapFetchWithPayment(async () => paymentRequiredResponse(), client);
    const pending = paid("https://pay.example/abort", {
      method: "POST",
      body: "abc",
      signal: controller.signal,
    });
    controller.abort();
    await expect(pending).rejects.toThrow(/aborted|AbortError|Failed to create payment payload/i);
    expect(signed).toBe(false);
  });

  it("rejects a body read aborted after it has started and cancels that source", async () => {
    let pulls = 0;
    let cancels = 0;
    const stream = new ReadableStream({
      pull(controller) {
        pulls += 1;
        if (pulls === 1) {
          controller.enqueue(new TextEncoder().encode("partial"));
          return;
        }
        return new Promise(() => undefined);
      },
      cancel() {
        cancels += 1;
      },
    });
    const controller = new AbortController();
    const request = new Request("https://pay.example/mid-abort", {
      method: "POST",
      body: stream,
      duplex: "half",
      signal: controller.signal,
    } as RequestInit);
    const call = httpPaymentCall(request, () => request);
    const pending = call.call.request.getBody?.();
    const started = Date.now();
    while (pulls < 2) {
      if (Date.now() - started > 1000) throw new Error("timed out waiting for the hung pull");
      await new Promise(resolve => setTimeout(resolve, 5));
    }
    controller.abort();
    const outcome = await Promise.race([
      pending?.then(
        () => "resolved",
        () => "rejected",
      ),
      new Promise<string>(resolve => setTimeout(() => resolve("pending"), 400)),
    ]);
    expect(outcome).toBe("rejected");
    expect(cancels).toBeGreaterThan(0);
    const again = await Promise.race([
      call.call.request.getBody?.().then(
        () => "resolved",
        () => "rejected",
      ),
      new Promise<string>(resolve => setTimeout(() => resolve("pending"), 200)),
    ]);
    expect(again).toBe("rejected");
  });

  it("aborts a wrapper body read without hanging or signing", async () => {
    let pulls = 0;
    const stream = new ReadableStream({
      pull(controller) {
        pulls += 1;
        if (pulls === 1) {
          controller.enqueue(new TextEncoder().encode("partial"));
          return;
        }
        return new Promise(() => undefined);
      },
    });
    let signed = false;
    const scheme: SchemeNetworkClient = {
      scheme: "test-scheme",
      async createPaymentPayload(_version, _requirements, context) {
        if (context?.call?.transport !== "http" || !context.call.request.getBody) {
          throw new Error("request_context_missing");
        }
        await context.call.request.getBody();
        signed = true;
        return { x402Version: 2, payload: { signature: "scheme_signature" } };
      },
    };
    const client = new x402Client().register(network, scheme).setSpendControls(false);
    const controller = new AbortController();
    const paid = wrapFetchWithPayment(async () => paymentRequiredResponse(), client);
    const pending = paid("https://pay.example/wrapper-abort", {
      method: "POST",
      body: stream,
      duplex: "half",
      signal: controller.signal,
    } as RequestInit);
    const started = Date.now();
    while (pulls < 2) {
      if (Date.now() - started > 1000) throw new Error("timed out waiting for the wrapper read");
      await new Promise(resolve => setTimeout(resolve, 5));
    }
    controller.abort();
    const outcome = await Promise.race([
      pending.then(
        () => "resolved",
        () => "rejected",
      ),
      new Promise<string>(resolve => setTimeout(() => resolve("pending"), 400)),
    ]);
    expect(outcome).toBe("rejected");
    expect(signed).toBe(false);
  });

  it("does not open a body-reader clone when no scheme reads the body", async () => {
    const originalClone = Request.prototype.clone;
    let clones = 0;
    Request.prototype.clone = function cloneRequest(this: Request) {
      clones += 1;
      return originalClone.call(this);
    };
    try {
      const scheme: SchemeNetworkClient = {
        scheme: "test-scheme",
        async createPaymentPayload() {
          return { x402Version: 2, payload: { signature: "scheme_signature" } };
        },
      };
      const client = new x402Client().register(network, scheme).setSpendControls(false);
      const paid = wrapFetchWithPayment(async request => {
        const current = request instanceof Request ? request : new Request(request);
        if (!current.headers.get("PAYMENT-SIGNATURE")) return paymentRequiredResponse();
        return new Response("ok", { status: 200 });
      }, client);
      clones = 0;
      const response = await paid("https://pay.example/unread", {
        method: "POST",
        body: "payload-bytes",
      });
      expect(response.status).toBe(200);
      expect(clones).toBe(2);

      clones = 0;
      const plain = wrapFetchWithPayment(async () => new Response("free", { status: 200 }), client);
      const free = await plain("https://pay.example/free", {
        method: "POST",
        body: "payload-bytes",
      });
      expect(free.status).toBe(200);
      expect(clones).toBe(1);

      clones = 0;
      const refusing = new x402Client()
        .register(network, {
          scheme: "test-scheme",
          async createPaymentPayload() {
            throw new Error("refused");
          },
        })
        .setSpendControls(false);
      const refused = wrapFetchWithPayment(async () => paymentRequiredResponse(), refusing);
      await expect(
        refused("https://pay.example/refused", { method: "POST", body: "x" }),
      ).rejects.toThrow("refused");
      expect(clones).toBe(1);
    } finally {
      Request.prototype.clone = originalClone;
    }
  });
});
